    82→/* ── Interactive 3D Shield (Landing Page) ── */
    83→function InteractiveShield() {
    84→  const containerRef = useRef<HTMLDivElement>(null);
    85→  const [mousePos, setMousePos] = useState({ x: 0.5, y: 0.5 });
    86→  const [isHovered, setIsHovered] = useState(false);
    87→  const [clickCount, setClickCount] = useState(0);
    88→
    89→  const handleMouseMove = useCallback((e: React.MouseEvent<HTMLDivElement>) => {
    90→    if (!containerRef.current) return;
    91→    const rect = containerRef.current.getBoundingClientRect();
    92→    setMousePos({ x: (e.clientX - rect.left) / rect.width, y: (e.clientY - rect.top) / rect.height });
    93→  }, []);
    94→
    95→  const handleMouseLeave = useCallback(() => {
    96→    setMousePos({ x: 0.5, y: 0.5 });
    97→    setIsHovered(false);
    98→  }, []);
    99→
   100→  const handleMouseEnter = useCallback(() => setIsHovered(true), []);
   101→
   102→  const handleClick = useCallback(() => setClickCount((c) => c + 1), []);
   103→
   104→  const offsetX = (mousePos.x - 0.5) * 16;
   105→  const offsetY = (mousePos.y - 0.5) * 10;
   106→
   107→  return (
   108→    <div ref={containerRef} onMouseMove={handleMouseMove} onMouseLeave={handleMouseLeave} onMouseEnter={handleMouseEnter}
   109→      onClick={handleClick}
   110→      className="relative w-full h-full flex items-center justify-center cursor-pointer select-none"
   111→      style={{ perspective: '800px' }}>
   112→      <div className="absolute inset-0 pointer-events-none" style={{
   113→        background: `radial-gradient(ellipse 60% 50% at 50% 45%, rgba(37, 99, 235, ${isHovered ? 0.15 : 0.08}) 0%, transparent 70%)`,
   114→        transition: 'background 0.4s ease',
   115→      }} />
   116→      <motion.div
   117→        className="relative"
   118→        initial={{ opacity: 0, scale: 0.85 }}
   119→        animate={{
   120→          opacity: 1,
   121→          scale: isHovered ? 1.06 : (clickCount > 0 ? [1, 1.08, 1] : 1),
   122→          y: [0, -6, 0],
   123→        }}
   124→        transition={{
   125→          opacity: { duration: 0.8 },
   126→          scale: isHovered ? { duration: 0.3, ease: 'easeOut' } : { duration: 0.4, ease: 'easeInOut' },
   127→          y: { duration: 5, repeat: Infinity, ease: 'easeInOut' },
   128→        }}
   129→        style={{ transformStyle: 'preserve-3d', transform: `rotateY(${offsetX * 1}deg) rotateX(${offsetY * -0.8}deg)`, transition: 'transform 0.15s ease-out' }}>
   130→        {/* Outer ring */}
   131→        <motion.div className="absolute" style={{ transform: 'translateZ(20px)', top: '-24px', left: '-24px', right: '-24px', bottom: '-24px' }}
   132→          animate={{ opacity: isHovered ? 0.4 : 0.2 }} transition={{ duration: 0.3 }}>
   133→          <svg viewBox="0 0 240 280" className="w-full h-full">
   134→            <path d="M120 16 L220 60 L220 180 C220 220 180 260 120 270 C60 260 20 220 20 180 L20 60 Z" fill="none" stroke="#2563EB" strokeWidth="1.5" />
   135→            <circle cx="120" cy="140" r="110" fill="none" stroke="#2563EB" strokeWidth="0.5" opacity="0.3" strokeDasharray="4 8" />
   136→          </svg>
   137→        </motion.div>
   138→        {/* Mid ring */}
   139→        <motion.div className="absolute" style={{ transform: 'translateZ(10px)', top: '-12px', left: '-12px', right: '-12px', bottom: '-12px' }}
   140→          animate={{ opacity: isHovered ? 0.5 : 0.3 }} transition={{ duration: 0.3 }}>
   141→          <svg viewBox="0 0 216 256" className="w-full h-full">
   142→            <path d="M108 20 L196 56 L196 168 C196 200 160 236 108 244 C56 236 20 200 20 168 L20 56 Z" fill="none" stroke="#2563EB" strokeWidth="2" strokeDasharray="6 4" />
   143→          </svg>
   144→        </motion.div>
   145→        {/* Main shield */}
   146→        <motion.div style={{ transform: 'translateZ(0px)' }}
   147→          animate={{ scale: clickCount > 0 ? [1, 1.05, 1] : 1 }}
   148→          transition={{ duration: 0.4 }}>
   149→          <svg viewBox="0 0 192 232" width="192" height="232" className="drop-shadow-lg">
   150→            <defs>
   151→              <linearGradient id="shieldGrad" x1="0%" y1="0%" x2="100%" y2="100%">
   152→                <stop offset="0%" stopColor="#2563EB" stopOpacity={isHovered ? 0.25 : 0.15} />
   153→                <stop offset="50%" stopColor="#2563EB" stopOpacity={isHovered ? 0.15 : 0.08} />
   154→                <stop offset="100%" stopColor="#1E40AF" stopOpacity={isHovered ? 0.2 : 0.12} />
   155→              </linearGradient>
   156→              <linearGradient id="shieldStroke" x1="0%" y1="0%" x2="100%" y2="100%">
   157→                <stop offset="0%" stopColor="#2563EB" />
   158→                <stop offset="100%" stopColor="#1E40AF" />
   159→              </linearGradient>
   160→            </defs>
   161→            <path d="M96 12 L180 48 L180 148 C180 178 148 210 96 220 C44 210 12 178 12 148 L12 48 Z" fill="url(#shieldGrad)" stroke="url(#shieldStroke)" strokeWidth="2.5" />
   162→            <path d="M96 32 L164 60 L164 140 C164 165 136 192 96 200 C56 192 28 165 28 140 L28 60 Z" fill="none" stroke="#2563EB" strokeWidth="1" opacity="0.4" />
   163→            <rect x="78" y="105" width="36" height="30" rx="4" fill="#2563EB" opacity="0.9" />
   164→            <path d="M86 105 L86 92 C86 82 96 74 106 82 L106 92 L106 105" fill="none" stroke="#2563EB" strokeWidth="2.5" strokeLinecap="round" opacity="0.9" />
   165→            <circle cx="96" cy="118" r="4" fill="#FFFFFF" />
   166→            <path d="M80 56 L92 68 L116 44" fill="none" stroke="#16A34A" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
   167→          </svg>
   168→        </motion.div>
   169→        {/* Floating particles - interactive */}
   170→        {[0, 1, 2, 3, 4].map((i) => (
   171→          <motion.div key={i} className="absolute w-2 h-2 rounded-full"
   172→            style={{
   173→              left: `${45 + (i % 3) * 10}%`,
   174→              top: `${30 + i * 10}%`,
   175→              backgroundColor: isHovered ? '#2563EB' : 'rgba(37,99,235,0.3)',
   176→            }}
   177→            animate={{
   178→              opacity: isHovered ? [0.4, 0.9, 0.4] : [0, 0.6, 0],
   179→              y: [0, -(20 + i * 12)],
   180→              x: [(i % 2 === 0 ? -1 : 1) * (8 + i * 4)],
   181→              scale: isHovered ? [1, 1.5, 1] : [1, 1, 1],
   182→            }}
   183→            transition={{ duration: isHovered ? 2 : 3 + i * 0.5, repeat: Infinity, delay: i * 0.6, ease: 'easeInOut' }}
   184→          />
   185→        ))}
   186→        {/* Hover glow ring */}
   187→        <AnimatePresence>
   188→          {isHovered && (
   189→            <motion.div
   190→              initial={{ opacity: 0, scale: 0.8 }}
   191→              animate={{ opacity: 0.15, scale: 1.2 }}
   192→              exit={{ opacity: 0, scale: 0.8 }}
   193→              className="absolute inset-0 rounded-full pointer-events-none"
   194→              style={{ background: 'radial-gradient(circle, rgba(37,99,235,0.2) 0%, transparent 70%)' }}
   195→            />
   196→          )}
   197→        </AnimatePresence>
   198→      </motion.div>
   199→    </div>
   200→  );
   201→}
