   261→/* ── Spotlight Heading (Hero Text Interaction) ── */
   262→function SpotlightHeading() {
   263→  const containerRef = useRef<HTMLDivElement>(null);
   264→  const mouseX = useMotionValue(0);
   265→  const mouseY = useMotionValue(0);
   266→  const [isHovered, setIsHovered] = useState(false);
   267→
   268→  const smoothX = useSpring(mouseX, { stiffness: 300, damping: 30, mass: 0.5 });
   269→  const smoothY = useSpring(mouseY, { stiffness: 300, damping: 30, mass: 0.5 });
   270→
   271→  const maskImage = useTransform(
   272→    [smoothX, smoothY] as const,
   273→    ([x, y]) =>
   274→      `radial-gradient(circle 120px at ${x}px ${y}px, white 0%, rgba(255,255,255,0.6) 40%, transparent 70%)`
   275→  );
   276→
   277→  const spotLightBg = useTransform(
   278→    [smoothX, smoothY] as const,
   279→    ([x, y]) =>
   280→      `radial-gradient(circle 200px at ${x}px ${y}px, rgba(37,99,235,0.12) 0%, rgba(37,99,235,0.04) 40%, transparent 70%)`
   281→  );
   282→
   283→  const handleMouseMove = useCallback((e: React.MouseEvent<HTMLDivElement>) => {
   284→    if (!containerRef.current) return;
   285→    const rect = containerRef.current.getBoundingClientRect();
   286→    mouseX.set(e.clientX - rect.left);
   287→    mouseY.set(e.clientY - rect.top);
   288→  }, [mouseX, mouseY]);
   289→
   290→  const handleMouseEnter = useCallback(() => setIsHovered(true), []);
   291→  const handleMouseLeave = useCallback(() => setIsHovered(false), []);
   292→
   293→  const headingText = "The future of banking\nauthentication";
   294→  const headingLines = headingText.split('\n');
   295→
   296→  const headingClasses = "font-hero text-4xl md:text-5xl lg:text-[3.5rem] font-extrabold tracking-tight leading-[1.1]";
   297→
   298→  return (
   299→    <motion.div
   300→      ref={containerRef}
   301→      variants={fadeInUp}
   302→      onMouseMove={handleMouseMove}
   303→      onMouseEnter={handleMouseEnter}
   304→      onMouseLeave={handleMouseLeave}
   305→      className="relative mb-4 cursor-default select-none overflow-hidden rounded-lg"
   306→    >
   307→      {/* Base text layer — dark navy */}
   308→      <h1 className={`${headingClasses} text-[#0F172A]`}>
   309→        {headingLines.map((line, i) => (
   310→          <span key={i}>
   311→            {line}
   312→            {i < headingLines.length - 1 && <br />}
   313→          </span>
   314→        ))}
   315→      </h1>
   316→
   317→      {/* Spotlight glow — blue radial gradient following cursor */}
   318→      <motion.div
   319→        className="absolute inset-0 pointer-events-none"
   320→        style={{
   321→          background: spotLightBg,
   322→          filter: 'blur(30px)',
   323→          opacity: isHovered ? 1 : 0,
   324→          transition: 'opacity 0.4s ease',
   325→          willChange: 'background',
   326→        }}
   327→      />
   328→
   329→      {/* Revealed text layer — white, masked by cursor spotlight */}
   330→      <motion.h1
   331→        className={`absolute inset-0 ${headingClasses} text-white`}
   332→        style={{
   333→          WebkitMaskImage: maskImage,
   334→          maskImage: maskImage,
   335→          willChange: 'mask-image, -webkit-mask-image',
   336→        }}
   337→      >
   338→        {headingLines.map((line, i) => (
   339→          <span key={i}>
   340→            {line}
   341→            {i < headingLines.length - 1 && <br />}
   342→          </span>
   343→        ))}
   344→      </motion.h1>
   345→    </motion.div>
   346→  );
   347→}
