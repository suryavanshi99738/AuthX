   261→   261→/* ── Spotlight Heading (Hero Text Interaction) ── */
   262→   262→function SpotlightHeading() {
   263→   263→  const containerRef = useRef<HTMLDivElement>(null);
   264→   264→  const mouseX = useMotionValue(0);
   265→   265→  const mouseY = useMotionValue(0);
   266→   266→  const [isHovered, setIsHovered] = useState(false);
   267→   267→
   268→   268→  const smoothX = useSpring(mouseX, { stiffness: 300, damping: 30, mass: 0.5 });
   269→   269→  const smoothY = useSpring(mouseY, { stiffness: 300, damping: 30, mass: 0.5 });
   270→   270→
   271→   271→  const maskImage = useTransform(
   272→   272→    [smoothX, smoothY] as const,
   273→   273→    ([x, y]) =>
   274→   274→      `radial-gradient(circle 120px at ${x}px ${y}px, white 0%, rgba(255,255,255,0.6) 40%, transparent 70%)`
   275→   275→  );
   276→   276→
   277→   277→  const spotLightBg = useTransform(
   278→   278→    [smoothX, smoothY] as const,
   279→   279→    ([x, y]) =>
   280→   280→      `radial-gradient(circle 200px at ${x}px ${y}px, rgba(37,99,235,0.12) 0%, rgba(37,99,235,0.04) 40%, transparent 70%)`
   281→   281→  );
   282→   282→
   283→   283→  const handleMouseMove = useCallback((e: React.MouseEvent<HTMLDivElement>) => {
   284→   284→    if (!containerRef.current) return;
   285→   285→    const rect = containerRef.current.getBoundingClientRect();
   286→   286→    mouseX.set(e.clientX - rect.left);
   287→   287→    mouseY.set(e.clientY - rect.top);
   288→   288→  }, [mouseX, mouseY]);
   289→   289→
   290→   290→  const handleMouseEnter = useCallback(() => setIsHovered(true), []);
   291→   291→  const handleMouseLeave = useCallback(() => setIsHovered(false), []);
   292→   292→
   293→   293→  const headingText = "The future of banking\nauthentication";
   294→   294→  const headingLines = headingText.split('\n');
   295→   295→
   296→   296→  const headingClasses = "font-hero text-4xl md:text-5xl lg:text-[3.5rem] font-extrabold tracking-tight leading-[1.1]";
   297→   297→
   298→   298→  return (
   299→   299→    <motion.div
   300→   300→      ref={containerRef}
   301→   301→      variants={fadeInUp}
   302→   302→      onMouseMove={handleMouseMove}
   303→   303→      onMouseEnter={handleMouseEnter}
   304→   304→      onMouseLeave={handleMouseLeave}
   305→   305→      className="relative mb-4 cursor-default select-none overflow-hidden rounded-lg"

   309→   309→        {headingLines.map((line, i) => (
   310→   310→          <span key={i}>
   311→   311→            {line}
   312→   312→            {i < headingLines.length - 1 && <br />}
   313→   313→          </span>
   314→   314→        ))}
   315→   315→      </h1>
   316→   316→
   317→   317→      {/* Spotlight glow — blue radial gradient following cursor */}
   318→   318→      <motion.div
   319→   319→        className="absolute inset-0 pointer-events-none"
   320→   320→        style={{
   321→   321→          background: spotLightBg,
   322→   322→          filter: 'blur(30px)',
   323→   323→          opacity: isHovered ? 1 : 0,
   324→   324→          transition: 'opacity 0.4s ease',
   325→   325→          willChange: 'background',
   326→   326→        }}
   327→   327→      />
   328→   328→
   329→   329→      {/* Revealed text layer — white, masked by cursor spotlight */}
   330→   330→      <motion.h1
   331→   331→        className={`absolute inset-0 ${headingClasses} text-white`}
   332→   332→        style={{
   333→   333→          WebkitMaskImage: maskImage,
   334→   334→          maskImage: maskImage,
   335→   335→          willChange: 'mask-image, -webkit-mask-image',
   336→   336→        }}
   337→   337→      >
   338→   338→        {headingLines.map((line, i) => (
   339→   339→          <span key={i}>
   340→   340→            {line}
   341→   341→            {i < headingLines.length - 1 && <br />}
   342→   342→          </span>
   343→   343→        ))}
   344→   344→      </motion.h1>
   345→   345→    </motion.div>
   346→   346→  );
   347→   347→}
